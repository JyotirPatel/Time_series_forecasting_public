3. **Exact config/code changes needed**
   Add a new predictor class in `07_BASELINE_MODEL_IMPLEMENTATIONS__baselines.py`, modeled on the existing `LinearBlendRegressor` plus the rule-matching style already used in `WarmStartLastTargetBlendRegressor`.

```python
@dataclass
class ConditionalLinearBlendRegressor:
    default_predictor: BatchPredictor
    slice_predictor: BatchPredictor
    blend_rules: list[dict[str, object]]   # each rule has conditions + weight in [0,1]

    def fit(self, train_frame: pd.DataFrame) -> "ConditionalLinearBlendRegressor":
        self.default_predictor.fit(train_frame)
        self.slice_predictor.fit(train_frame)
        return self

    def predict_batch(self, batch_frame: pd.DataFrame) -> np.ndarray:
        default_pred = np.asarray(self.default_predictor.predict_batch(batch_frame), dtype=float)
        slice_pred = np.asarray(self.slice_predictor.predict_batch(batch_frame), dtype=float)
        out = default_pred.copy()
        for rule in self.blend_rules:
            w = float(rule["weight"])
            mask = np.ones(len(batch_frame), dtype=bool)
            for col, val in rule.items():
                if col == "weight":
                    continue
                mask &= batch_frame[col].to_numpy() == val
            out[mask] = (1.0 - w) * default_pred[mask] + w * slice_pred[mask]
        return out
```
